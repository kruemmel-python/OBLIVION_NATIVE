project(OblivionNative VERSION 2.1.1 LANGUAGES CXX RC)

  message(FATAL_ERROR "Oblivion Native targets Windows only.")

target_link_libraries(oblivion_core PUBLIC bcrypt crypt32 shell32 ole32 oleaut32 wbemuuid rstrtmgr comctl32 advapi32 uxtheme dwmapi)

function(strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /sdl /guard:cf)
    get_target_property(target_type ${target} TYPE)
    if(NOT target_type STREQUAL "STATIC_LIBRARY")
      target_link_options(${target} PRIVATE /WX /guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
    endif()
  endif()
endfunction()
foreach(t oblivion_core oblivion oblivion_cli oblivion_device oblivion_shell)
  strict(${t})
endforeach()
